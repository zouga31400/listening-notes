# Playlists by decade, genre and mood, created automatically on first use.

# service (the original year is used, not the date of a remaster); {period} is

decade_playlist = Radio {decade}

mood_playlist =
mood_period_playlist = Radio {mood} {period}

# Moods, measured on a 30-second preview of each track: energy (0-100) mixes rhythmic
# intensity, loudness and brightness. Calm up to calm_max_energy (or under 80 BPM
# without high energy), energetic from energetic_min_energy, groove in between.
mood_names = Calme, Groove, Énergique
# Thresholds calibrated on 46 tracks (11/11/2026): about a third of the tracks in each mood.
calm_max_energy = 53
energetic_min_energy = 63

# Notion database for tracks recognized but missing from the service.
# Database ID (not secret). The token goes in the NOTION_TOKEN GitHub secret.
notion_missing_db = 0cfa728c14294956be448e18eb460be1

# Notion history database: every recognized track with its estimated start time.
notion_history_db = 1099479c43a944e0a358727716a0d49b

# Time listening starts, Paris time (HH:MM). The job may be started a little earlier;
# the script then waits until this time.
listen_start = 21:00

# End of the show. The show ends at a different time every night: after
# end_detection_from, when end_streak tracks are played in a row without speech,
# the show is considered over. The first track of that run is kept, the next ones
# are ignored, listening stops and a "show ended" signal is sent through ntfy.
# Listening stops at latest_end in any case (the job is limited to 6 hours).
end_detection_from = 00:00
end_streak = 3
latest_end = 02:00

# Listening frequency, in seconds (minimum 30).
# listen_interval: normal pace, right after a track.
# speech_interval: pace after 4 tries without music.
# A track longer than 2 minutes is always caught at 90 s.
listen_interval = 45
speech_interval = 90

# Number of matching recognitions before a track is added (1 = at the first one).
# 2 avoids false matches when the host talks over the music; a track about to end
# is added at once.
confirmations = 2

# Daily playlists older than this many days are deleted at the end of the evening
# (monthly playlists are kept). 0 = never delete.
keep_daily_days = 30
