# Time listening starts, Paris time (HH:MM). The job may be started a little earlier;
# the script then waits until this time.
listen_start = 21:00

# End of the show. The show ends at a different time every night: after
# end_detection_from, when end_streak tracks are played in a row without speech,
# the show is considered over just before that run: all its tracks are ignored,
# listening stops and a "show ended" signal is sent through ntfy. A short
# announcement between two tracks counts as speech. After end_detection_from, a
# track is added (and notified) once speech is heard after it.
# Listening stops at latest_end in any case (the job is limited to 6 hours).
end_detection_from = 00:00
end_streak = 3
latest_end = 02:00

# Start of the show: the evening begins with music played back to back, which is not
# the show. Tracks are ignored until speech is heard; the show starts with the first
# track after speech. If no speech is heard before show_start_until (Paris time), the
# show is taken as started. "off" keeps every track from the start.
show_start_until = 21:30

# Short music excerpts played during the show (heard only once, between speech): they
# go only to this playlist (created if missing) and to the Notion history with the
# status "Excerpt", not to the other playlists, and without a notification.
# Empty = only in the history.

# Listening frequency, in seconds (minimum 30).
# listen_interval: normal pace, right after a track.
# speech_interval: pace after 4 tries without music.
